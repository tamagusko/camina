    ```python
    """Lightweight telemetry readers for the sensor daemon.

    All functions are cross-platform safe: they return None / a sentinel
    string when the underlying command is unavailable (e.g. non-Pi dev host,
    macOS). This lets the same heartbeat path run in CI without branching.
    """
    from __future__ import annotations

    import logging
    import resource
    import shutil
    import subprocess
    from pathlib import Path
    from typing import Optional

    logger = logging.getLogger(__name__)


    def read_throttled() -> Optional[str]:
        """Return vcgencmd get_throttled output (e.g. '0x0' or '0x50005').

        Returns None on hosts without vcgencmd (macOS, non-Pi Linux).
        """
        if shutil.which("vcgencmd") is None:
            return None
        try:
            out = subprocess.run(
                ["vcgencmd", "get_throttled"],
                capture_output=True, text=True, timeout=2, check=True,
            ).stdout.strip()
            return out.split("=", 1)[1] if "=" in out else out
        except (subprocess.CalledProcessError, subprocess.TimeoutExpired) as exc:
            logger.debug("vcgencmd get_throttled failed: %s", exc)
            return None


    def read_rss_mb() -> Optional[float]:
        """Process RSS in MB via resource.getrusage; cross-platform."""
        try:
            ru = resource.getrusage(resource.RUSAGE_SELF).ru_maxrss
        except (OSError, AttributeError):
            return None
        # ru > 2^30 → bytes (macOS); else KB (Linux).
        return round((ru / (1024 * 1024)) if ru > 2**30 else (ru / 1024), 1)


    def read_disk_free_mb(path: Path = Path("/var/lib/camina")) -> Optional[float]:
        """Bytes-free on the partition containing `path`, in MB.

        Returns None if the path does not exist (e.g. non-Pi dev host without
        the /var/lib/camina mount).
        """
        if not path.exists():
            # Try current directory as fallback for dev hosts.
            path = Path.cwd()
        try:
            usage = shutil.disk_usage(str(path))
            return round(usage.free / (1024 * 1024), 1)
        except OSError as exc:
            logger.debug("disk_usage(%s) failed: %s", path, exc)
            return None


    __all__ = ["read_throttled", "read_rss_mb", "read_disk_free_mb"]
    ```

    Exact structure:

    ```python
    """48-hour soak monitor — evaluates EDGE-08 gates against collected heartbeats.

    Usage:
        uv run python scripts/soak_monitor.py \\
            --heartbeats-file ./heartbeats.jsonl \\
            --rss-tolerance 0.10 \\
            --expected-sensor-id D01 \\
            --out docs/benchmarks/soak-20260425-48h.md

    PLAN-CHECK 01-04 monitor-sanity flag: --expected-sensor-id sanity check
    catches the case where a stale/wrong/cross-sensor JSONL file is used.
    Also detects first-vs-last sensor_id drift even without the flag.
    """
    from __future__ import annotations

    import argparse
    import json
    import logging
    import sys
    from dataclasses import dataclass, field
    from datetime import datetime, timezone
    from pathlib import Path

    logger = logging.getLogger(__name__)


    SOAK_GATES = ("rss_flat", "no_restarts", "no_throttle_bits")


    @dataclass
    class GateResult:
        name: str
        passed: bool
        reason: str = ""


    @dataclass
    class SoakReport:
        sensor_id: str
        started_at: str
        ended_at: str
        heartbeat_count: int
        rss_min: float | None
        rss_max: float | None
        rss_mean: float | None
        uptime_max_s: int
        throttled_nonzero_count: int
        sensor_id_mismatch_count: int = 0
        gates: list[GateResult] = field(default_factory=list)

        @property
        def passed(self) -> bool:
            return all(g.passed for g in self.gates)


    def load_heartbeats(path: Path) -> list[dict]:
        items = []
        with path.open() as f:
            for line in f:
                line = line.strip()
                if not line:
                    continue
                items.append(json.loads(line))
        return items


    def evaluate_rss_gate(heartbeats: list[dict], tolerance: float = 0.10) -> GateResult:
        rss_values = [hb["rss_mb"] for hb in heartbeats if hb.get("rss_mb") is not None]
        if len(rss_values) < 2:
            return GateResult("rss_flat", False, "insufficient_rss_samples")
        lo, hi = min(rss_values), max(rss_values)
        growth = (hi - lo) / lo if lo > 0 else 0.0
        if growth > tolerance:
            return GateResult("rss_flat", False,
                              f"rss_growth_{int(growth*100)}pct_exceeds_{int(tolerance*100)}pct_tolerance")
        return GateResult("rss_flat", True)


    def evaluate_throttle_gate(heartbeats: list[dict]) -> GateResult:
        nonzero = [hb for hb in heartbeats
                   if hb.get("throttled") and hb["throttled"] != "0x0"]
        if nonzero:
            return GateResult("no_throttle_bits", False,
                              f"{len(nonzero)}_heartbeats_with_throttle_set")
        return GateResult("no_throttle_bits", True)


    def evaluate_restart_gate(heartbeats: list[dict]) -> GateResult:
        uptimes = [hb.get("uptime_s", 0) for hb in heartbeats]
        for prev, curr in zip(uptimes, uptimes[1:]):
            if curr < prev:
                return GateResult("no_restarts", False,
                                  f"uptime_dropped_from_{prev}_to_{curr}")
        return GateResult("no_restarts", True)


    def check_sensor_id_consistency(
        heartbeats: list[dict],
        expected: str | None = None,
    ) -> int:
        """Return count of mismatched heartbeats; emit warnings.

        PLAN-CHECK 01-04 monitor-sanity flag:
        - If `expected` is set, warn for each heartbeat whose sensor_id != expected.
        - If first/last differ regardless of `expected`, warn once.
        Returns the total mismatch count.
        """
        if not heartbeats:
            return 0
        first_id = heartbeats[0].get("sensor_id")
        last_id = heartbeats[-1].get("sensor_id")
        if first_id != last_id:
            logger.warning(
                "first/last heartbeat sensor_id differ: %s vs %s — likely stale or merged JSONL",
                first_id, last_id,
            )
        if expected is None:
            return 0
        mismatches = [hb for hb in heartbeats if hb.get("sensor_id") != expected]
        for hb in mismatches:
            logger.warning(
                "sensor_id mismatch: expected=%s observed=%s ts=%s",
                expected, hb.get("sensor_id"), hb.get("ts"),
            )
        return len(mismatches)


    def build_report(
        heartbeats: list[dict],
        tolerance: float,
        expected_sensor_id: str | None = None,
    ) -> SoakReport:
        sensor_id = (heartbeats[0].get("sensor_id") if heartbeats else "unknown")
        started_at = (heartbeats[0].get("ts") if heartbeats else "")
        ended_at = (heartbeats[-1].get("ts") if heartbeats else "")
        rss_values = [hb["rss_mb"] for hb in heartbeats if hb.get("rss_mb") is not None]
        rss_min = min(rss_values) if rss_values else None
        rss_max = max(rss_values) if rss_values else None
        rss_mean = (sum(rss_values) / len(rss_values)) if rss_values else None
        uptime_max_s = max((hb.get("uptime_s", 0) for hb in heartbeats), default=0)
        throttled_nonzero_count = len([hb for hb in heartbeats
                                       if hb.get("throttled") and hb["throttled"] != "0x0"])
        sensor_id_mismatch_count = check_sensor_id_consistency(heartbeats, expected_sensor_id)
        gates = [
            evaluate_rss_gate(heartbeats, tolerance),
            evaluate_restart_gate(heartbeats),
            evaluate_throttle_gate(heartbeats),
        ]
        return SoakReport(
            sensor_id=sensor_id,
            started_at=started_at,
            ended_at=ended_at,
            heartbeat_count=len(heartbeats),
            rss_min=rss_min,
            rss_max=rss_max,
            rss_mean=rss_mean,
            uptime_max_s=uptime_max_s,
            throttled_nonzero_count=throttled_nonzero_count,
            sensor_id_mismatch_count=sensor_id_mismatch_count,
            gates=gates,
        )


    def render_report(r: SoakReport) -> str:
        verdict = lambda g: "PASS" if g.passed else "FAIL"  # noqa: E731
        lines = [
            f"# Soak report — {r.sensor_id} — {r.started_at} → {r.ended_at}",
            "",
            f"Heartbeats collected: {r.heartbeat_count}",
            "",
            "## Gate evaluation",
            "",
            "| Gate | Result | Reason |",
            "|---|---|---|",
        ]
        lines.extend(f"| {g.name} | {verdict(g)} | {g.reason or 'n/a'} |" for g in r.gates)
        lines.extend([
            "",
            f"**Overall: {'PASS' if r.passed else 'FAIL'}**",
            "",
            "## Statistics",
            "",
            "| Metric | Value |",
            "|---|---|",
            f"| Heartbeats | {r.heartbeat_count} |",
            f"| RSS min / mean / max (MB) | {r.rss_min} / {r.rss_mean} / {r.rss_max} |",
            f"| Max uptime (s) | {r.uptime_max_s} |",
            f"| Throttled heartbeats | {r.throttled_nonzero_count} |",
            f"| sensor_id mismatch count | {r.sensor_id_mismatch_count} |",
            "",
        ])
        return "\n".join(lines)


    def main(argv: list[str] | None = None) -> int:
        logging.basicConfig(level=logging.INFO,
                            format="%(asctime)s %(levelname)s %(name)s: %(message)s")
        parser = argparse.ArgumentParser(description="CAMINA 48-hour soak monitor")
        parser.add_argument("--heartbeats-file", type=Path, required=True)
        parser.add_argument("--rss-tolerance", type=float, default=0.10)
        parser.add_argument("--expected-sensor-id", type=str, default=None,
                            help="Warn on any heartbeat whose sensor_id differs from this value")
        parser.add_argument("--out", type=Path, required=True)
        args = parser.parse_args(argv)

        if not args.heartbeats_file.exists():
            logger.error("Heartbeats file not found: %s", args.heartbeats_file)
            return 2
        heartbeats = load_heartbeats(args.heartbeats_file)
        report = build_report(heartbeats, args.rss_tolerance, args.expected_sensor_id)
        args.out.parent.mkdir(parents=True, exist_ok=True)
        args.out.write_text(render_report(report))
        logger.info("Soak report written to %s (overall PASS=%s)", args.out, report.passed)
        return 0 if report.passed else 1


    if __name__ == "__main__":
        sys.exit(main())
    ```
