    ```python
    """systemd watchdog notifier.

    Wraps sd_notify("WATCHDOG=1") so the main loop can prove liveness to
    systemd via WatchdogSec. On non-systemd hosts (dev laptops, CI), becomes
    a no-op. Also wraps sd_notify("READY=1") for Type=notify boot semantics.

    PLAN-CHECK 01-03 flag: notify_ready() MUST fire IMMEDIATELY at the top of
    SensorDaemon.start() — before any other work — to stay within
    TimeoutStartSec (default 90s) when an integrity check or NTP poll could
    otherwise burn the budget.
    """
    from __future__ import annotations

    import logging
    import os
    import socket

    logger = logging.getLogger(__name__)

    _WATCHDOG_PAYLOAD = b"WATCHDOG=1"
    _READY_PAYLOAD = b"READY=1"


    def _send(payload: bytes) -> bool:
        sock_path = os.environ.get("NOTIFY_SOCKET")
        if not sock_path:
            return False
        # Abstract namespace sockets start with '@'; replace with '\0' per man sd_notify.
        addr = "\0" + sock_path[1:] if sock_path.startswith("@") else sock_path
        try:
            with socket.socket(socket.AF_UNIX, socket.SOCK_DGRAM) as s:
                s.sendto(payload, addr)
            return True
        except OSError as exc:
            logger.warning("sd_notify(%s) failed: %s", payload.decode(), exc)
            return False


    def notify_watchdog() -> bool:
        """Send WATCHDOG=1 to systemd via $NOTIFY_SOCKET.

        Returns True if the notification was sent; False if no systemd socket
        is present (dev host, CI, tests).
        """
        return _send(_WATCHDOG_PAYLOAD)


    def notify_ready() -> bool:
        """Send READY=1 to systemd (called once at startup, IMMEDIATELY)."""
        return _send(_READY_PAYLOAD)


    __all__ = ["notify_watchdog", "notify_ready"]
    ```

    **File 2: `src/camina/service/ntp_gate.py`** (new, ~50 lines).

    ```python
    """Boot-time NTP sync gate.

    The Raspberry Pi has no RTC. If systemd-timesyncd / chrony has not synced
    by the time the daemon starts, window_start timestamps will drift from
    the server clock and break the (sensor_id, window_start) idempotency
    contract (Pitfall 7 in PITFALLS.md).
    """
    from __future__ import annotations

    import logging
    import shutil
    import subprocess
    import time

    logger = logging.getLogger(__name__)


    def _query_timedatectl() -> str | None:
        """Return 'yes' / 'no' / None. None = timedatectl unavailable."""
        if shutil.which("timedatectl") is None:
            return None
        try:
            result = subprocess.run(
                ["timedatectl", "show", "--property=NTPSynchronized", "--value"],
                capture_output=True, text=True, timeout=5, check=True,
            )
            return result.stdout.strip()
        except (subprocess.CalledProcessError, subprocess.TimeoutExpired) as exc:
            logger.warning("timedatectl query failed: %s", exc)
            return None


    def wait_for_ntp_sync(timeout_s: float = 300.0, poll_s: float = 10.0) -> bool:
        """Block until NTP is synced or timeout.

        Returns True if NTPSynchronized=yes. False on timeout or if timedatectl
        is unavailable (non-Pi host).
        """
        deadline = time.monotonic() + timeout_s
        first = _query_timedatectl()
        if first is None:
            logger.error("timedatectl not available; cannot verify NTP sync")
            return False
        if first == "yes":
            logger.info("NTP already synced at startup")
            return True
        logger.warning("NTP not yet synced; polling every %.1fs up to %.0fs", poll_s, timeout_s)
        while time.monotonic() < deadline:
            time.sleep(poll_s)
            status = _query_timedatectl()
            if status == "yes":
                logger.info("NTP sync achieved")
                return True
        logger.error("NTP sync not achieved within %.0fs", timeout_s)
        return False


    __all__ = ["wait_for_ntp_sync"]
    ```

    ```python
    from __future__ import annotations

    import logging
    import sqlite3
    from datetime import datetime, timezone
    from pathlib import Path

    logger = logging.getLogger(__name__)


    def _integrity_check_or_quarantine(db_path: Path) -> None:
        """Run PRAGMA integrity_check; rename file on failure.

        If the file does not exist, does nothing (caller will create it).
        If integrity_check returns anything other than 'ok', the file is
        renamed to <db_path>.broken.<UTC-timestamp> and a fresh DB is created
        on the next connection. Callers should treat this as "outbox
        contents lost" and continue.
        """
        if not db_path.exists():
            return
        try:
            conn = sqlite3.connect(str(db_path))
            cur = conn.execute("PRAGMA integrity_check")
            result = cur.fetchone()
            conn.close()
        except sqlite3.DatabaseError as exc:
            logger.error("Failed to open %s for integrity_check: %s", db_path, exc)
            result = None

        if result is None or result[0] != "ok":
            ts = datetime.now(tz=timezone.utc).strftime("%Y%m%dT%H%M%SZ")
            broken = db_path.with_suffix(db_path.suffix + f".broken.{ts}")
            db_path.rename(broken)
            logger.error("Quarantined corrupt SQLite file: renamed %s -> %s", db_path, broken)


    __all__ = ["_integrity_check_or_quarantine"]
    ```
