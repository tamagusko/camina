    Runs the fine-tuned YOLO NCNN + Kalman tracker + WindowedCounter pipeline
    for a fixed duration, samples thermal/RSS/throttle metrics, and writes
    JSON + Markdown reports. Validates EDGE-03 (30-min in-enclosure benchmark

    docs/sensor_deployment.md §8.

    from dataclasses import asdict, dataclass, field
    from datetime import datetime, timezone
    from pathlib import Path
    from typing import Iterator

    import numpy as np

    from src.camina.service.compose import compose
    from src.camina.service.sensor_daemon import DaemonConfig, _read_cpu_temp

    logger = logging.getLogger(__name__)


    @dataclass
    class BenchSample:
        t_s: float                     # seconds since bench start
        fps_instant: float
        cpu_temp_c: float | None
        rss_mb: float

    @dataclass
    class BenchReport:
        sensor_id: str
        imgsz: int
        duration_s: float
        frames_total: int
        fps_mean: float
        fps_mean_1min: float           # rolling mean of last 60 s
        cpu_temp_max_c: float | None
        cpu_temp_mean_c: float | None
        rss_mb_max: float
        rss_mb_end: float
        throttled_at_start: str
        throttled_at_mid: str
        throttled_at_end: str
        samples: list[BenchSample] = field(default_factory=list)
        started_at: str = ""           # ISO8601 UTC
        ended_at: str = ""

    def _read_throttled() -> str:
        """Return vcgencmd get_throttled output (e.g. '0x0' or '0x50005').

        Returns '0xUNAVAILABLE' on non-Pi hosts.
        """
        if shutil.which("vcgencmd") is None:
            return "0xUNAVAILABLE"
        try:
            out = subprocess.run(
                ["vcgencmd", "get_throttled"],
                capture_output=True, text=True, timeout=2, check=True,
            ).stdout.strip()
            # Output: "throttled=0x0"
            return out.split("=", 1)[1] if "=" in out else out
        except (subprocess.CalledProcessError, subprocess.TimeoutExpired):
            return "0xERROR"

    def _read_rss_mb() -> float:
        """Process RSS in MB via resource.getrusage (Linux/macOS)."""
        # ru_maxrss is KB on Linux, bytes on macOS. Detect heuristically.
        ru = resource.getrusage(resource.RUSAGE_SELF).ru_maxrss
        # If ru > 2**30, assume bytes (macOS); else KB (Linux).
        return (ru / (1024 * 1024)) if ru > 2**30 else (ru / 1024)

    def _file_frame_source(video_path: Path, imgsz: int) -> Iterator[np.ndarray]:
        """Iterate RGB frames from a video file (dev/CI fallback).

        Bench-only; production daemon uses picamera2 RGB888 directly.
        """
        import cv2  # local import — only needed in bench, not in production daemon
        cap = cv2.VideoCapture(str(video_path))
        try:
            while True:
                ok, bgr = cap.read()
                if not ok:
                    cap.set(cv2.CAP_PROP_POS_FRAMES, 0)  # loop
                    continue
                rgb = cv2.cvtColor(bgr, cv2.COLOR_BGR2RGB)
                yield cv2.resize(rgb, (imgsz, imgsz))
        finally:
            cap.release()

    class _NoopPublisher:
        """Replaces HttpsPublisher during benchmark so no network I/O occurs."""

            self.counts_calls = 0
            self.daily_calls = 0
            self.heartbeat_calls = 0
        def post_counts(self, **_: object):
            self.counts_calls += 1
            from src.camina.io.https_publisher import PublisherResult
            return PublisherResult(delivered=True, enqueued=False, latest_config_version="")
        def post_daily(self, **_: object):
            self.daily_calls += 1

            self.heartbeat_calls += 1
            from src.camina.io.https_publisher import PublisherResult
            return PublisherResult(delivered=True, enqueued=False, latest_config_version="")

    def run_benchmark(
        cfg: DaemonConfig,
        duration_s: float,
        imgsz: int,
        video_path: Path | None,
        out_dir: Path,
    ) -> BenchReport:
        """Main benchmark loop. Returns a filled BenchReport."""
        # ... (implementation: build camera_factory per video_path, compose daemon,
        #      call daemon.replace_publisher(_NoopPublisher()), wrap detect_factory
        #      with a frame counter, run main loop until duration elapses, sample
        #      thermal/RSS every 5 s, capture throttle at start/mid/end, stop daemon,
        #      write reports)

    def main(argv: list[str] | None = None) -> int:
        logging.basicConfig(
            level=logging.INFO,
            format="%(asctime)s %(levelname)s %(name)s: %(message)s",
        )
        parser = argparse.ArgumentParser(description="CAMINA Pi 5 benchmark")
        parser.add_argument("--config", type=Path, required=True)
        parser.add_argument("--duration-min", type=float, default=30.0,
                            help="Benchmark duration in minutes (default 30)")
        parser.add_argument("--imgsz", type=int, default=480)
        parser.add_argument("--video", type=Path, default=None,
                            help="Use a video file instead of picamera2 (dev/CI)")
        parser.add_argument("--out-dir", type=Path, default=Path("docs/benchmarks"))
        args = parser.parse_args(argv)

        if not args.config.exists():
            logger.error("Config file not found: %s", args.config)
            return 2
        args.out_dir.mkdir(parents=True, exist_ok=True)

        cfg = DaemonConfig.from_yaml(args.config)
        duration_s = args.duration_min * 60.0
        report = run_benchmark(cfg, duration_s, args.imgsz, args.video, args.out_dir)

        # Write JSON
        ts = datetime.now(tz=timezone.utc).strftime("%Y%m%d-%H%M")
        json_path = args.out_dir / f"bench-{ts}.json"
        md_path = args.out_dir / f"bench-{ts}.md"
        json_path.write_text(json.dumps(asdict(report), indent=2, default=str))

        # Write Markdown (table with FPS, temp, RSS, throttle transcript)
        md_path.write_text(_render_markdown(report))

        logger.info("Benchmark complete. FPS mean=%.2f, 1-min rolling=%.2f, temp max=%s °C, throttled at end=%s",
                    report.fps_mean, report.fps_mean_1min,
                    report.cpu_temp_max_c, report.throttled_at_end)
        return 0

    if __name__ == "__main__":
        sys.exit(main())
