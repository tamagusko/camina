    ```python
    """Lightweight telemetry readers for the sensor daemon.

    All functions are cross-platform safe: they return None / a sentinel
    string when the underlying command is unavailable (e.g. non-Pi dev host,
    macOS). This lets the same heartbeat path run in CI without branching.
    """
    from __future__ import annotations

    import logging
    import resource
    import shutil
    import subprocess
    from pathlib import Path
    from typing import Optional

    logger = logging.getLogger(__name__)


    def read_throttled() -> Optional[str]:
        """Return vcgencmd get_throttled output (e.g. '0x0' or '0x50005').

        Returns None on hosts without vcgencmd (macOS, non-Pi Linux).
        """
        if shutil.which("vcgencmd") is None:
            return None
        try:
            out = subprocess.run(
                ["vcgencmd", "get_throttled"],
                capture_output=True, text=True, timeout=2, check=True,
            ).stdout.strip()
            return out.split("=", 1)[1] if "=" in out else out
        except (subprocess.CalledProcessError, subprocess.TimeoutExpired) as exc:
            logger.debug("vcgencmd get_throttled failed: %s", exc)
            return None


    def read_rss_mb() -> Optional[float]:
        """Process RSS in MB via resource.getrusage; cross-platform."""
        try:
            ru = resource.getrusage(resource.RUSAGE_SELF).ru_maxrss
        except (OSError, AttributeError):
            return None
        # ru > 2^30 → bytes (macOS); else KB (Linux).
        return round((ru / (1024 * 1024)) if ru > 2**30 else (ru / 1024), 1)


    def read_disk_free_mb(path: Path = Path("/var/lib/camina")) -> Optional[float]:
        """Bytes-free on the partition containing `path`, in MB.

        Returns None if the path does not exist (e.g. non-Pi dev host without
        the /var/lib/camina mount).
        """
        if not path.exists():
            # Try current directory as fallback for dev hosts.
            path = Path.cwd()
        try:
            usage = shutil.disk_usage(str(path))
            return round(usage.free / (1024 * 1024), 1)
        except OSError as exc:
            logger.debug("disk_usage(%s) failed: %s", path, exc)
            return None


    __all__ = ["read_throttled", "read_rss_mb", "read_disk_free_mb"]
    ```

    import logging
    import sys
    from dataclasses import dataclass, field
    from datetime import datetime, timezone
    from pathlib import Path

    logger = logging.getLogger(__name__)


    SOAK_GATES = ("rss_flat", "no_restarts", "no_throttle_bits")


    @dataclass
    class GateResult:
        name: str
        passed: bool
        reason: str = ""


    @dataclass
    class SoakReport:
        sensor_id: str
        started_at: str
        ended_at: str
        heartbeat_count: int
        rss_min: float | None
        rss_max: float | None
        rss_mean: float | None
        uptime_max_s: int
        throttled_nonzero_count: int
        gates: list[GateResult] = field(default_factory=list)

        @property
        def passed(self) -> bool:
            return all(g.passed for g in self.gates)


    def load_heartbeats(path: Path) -> list[dict]:
        items = []
        with path.open() as f:
            for line in f:
                line = line.strip()
                if not line:
                    continue
                items.append(json.loads(line))
        return items


    def evaluate_rss_gate(heartbeats: list[dict], tolerance: float = 0.10) -> GateResult:
        """PASS if max(rss_mb)/min(rss_mb) - 1 <= tolerance."""
        rss_values = [hb["rss_mb"] for hb in heartbeats if hb.get("rss_mb") is not None]
        if len(rss_values) < 2:
            return GateResult("rss_flat", False, "insufficient_rss_samples")
        lo, hi = min(rss_values), max(rss_values)
        growth = (hi - lo) / lo if lo > 0 else 0.0
        if growth > tolerance:
            return GateResult("rss_flat", False,
                              f"rss_growth_{int(growth*100)}pct_exceeds_{int(tolerance*100)}pct_tolerance")
        return GateResult("rss_flat", True)

        """PASS if no heartbeat has a non-zero throttled value."""
        nonzero = [hb for hb in heartbeats
                   if hb.get("throttled") and hb["throttled"] != "0x0"]
        if nonzero:
            return GateResult("no_throttle_bits", False,
                              f"{len(nonzero)}_heartbeats_with_throttle_set")
        return GateResult("no_throttle_bits", True)
